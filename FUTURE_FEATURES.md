## 🛠️ Как добавить новые функции

### Для ИИ моделей
```typescript
// Добавить нового провайдера в aiProviders.ts
export const NewAIProvider: AIProvider = {
  id: 'new-ai',
  name: 'New AI Model',
  apiUrl: 'https://api.newai.com',
  // ... остальная конфигурация
};
```

### Для автоматизации
```typescript
// Добавить новое правило в automationService.ts
{
  id: 'new-rule',
  name: '🎯 Новое правило',
  description: 'Описание функции',
  trigger: 'code_generation',
  actions: [{ type: 'custom_action' }],
  enabled: true,
  priority: 1
}
```

### Для генерации контента
```typescript
// Расширить imageService.ts
export class AudioService {

  }
}
```
